/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    // approach 1 : reverse the other half and start from starting from both half 
    int pairSum2(ListNode* head) {
        ListNode *slow;
        ListNode *fast;
        slow = head;
        fast = head;
        while(fast && fast->next){
            slow = slow->next;
            fast = fast->next->next;
        }
        ListNode *back = NULL;
        ListNode *forw = slow;
        while(forw!=NULL){
            forw = forw->next;
            slow->next = back;
            back = slow;
            slow = forw; 
        }
        ListNode *start = head;
        ListNode *end = back;
        int res = 0;
        while(end){
            res = max(res,(end->val + start->val));
            end = end->next;
            start = start->next;
        }
        
        return res;
    }
    //approach 2 : convert the link list to array and go from both end and compare 
    int pairSum(ListNode* head) {
        ListNode *pt;
        pt = head;
        vector<int> temp;
        while(pt!=NULL){
            temp.push_back(pt->val);
            pt = pt->next;
        }
        int res = 0;
        int n = temp.size();
        for(int i=0;i<n/2;i++){
            res = max(res,(temp[i]+temp[n-1-i]));
        }
        return res;
    }
};
