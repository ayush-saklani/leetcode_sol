/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left),
 * right(right) {}
 * };
 */
class Solution {
public:
    TreeNode* createBinaryTree(vector<vector<int>>& descriptions) {
        unordered_map<int, pair<TreeNode*,int>> temp; // { node_val , {node_ptr, is_it_someone's_child_? } }  
        TreeNode* root;
        for (int i = 0; i < descriptions.size(); i++) {
            // node hai toh nikalo ya toh naya create karo
            TreeNode* node;
            if(temp.count(descriptions[i][0])){
                node = temp[descriptions[i][0]].first;
            }else{
                node = new TreeNode(descriptions[i][0]);
                temp[descriptions[i][0]] = {node,false};
            }

            // child hai toh nikalo ya toh naya create karo
            TreeNode* child;
            if (temp.count(descriptions[i][1])) {
                child = temp[descriptions[i][1]].first;
                // child pehle se hai toh is someone child = true kardo
                temp[descriptions[i][1]].second = true;
            } else {
                child = new TreeNode(descriptions[i][1]);
                temp[descriptions[i][1]] = {child,true};
            }

            if (descriptions[i][2] == 0) { // acc to rules
                node->right = child;
            } else {
                node->left = child;
            }
        }
        // last mei ek hi bachega with is someone child? = false return that as answer
        for(auto &it : temp){
            if(it.second.second == false){
                root = it.second.first;
            }
        }
        return root;
    }
};
