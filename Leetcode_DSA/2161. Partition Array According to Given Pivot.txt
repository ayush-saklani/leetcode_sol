class Solution {
public:
    vector<int> pivotArray(vector<int>& nums, int pivot) {
        vector<int> prev;
        vector<int> equal;
        vector<int> next;
        for(int i=0;i<nums.size();i++){
            if(nums[i]<pivot){
                prev.push_back(nums[i]);
            }else if(nums[i]>pivot){
                next.push_back(nums[i]);
            }else if(nums[i] == pivot){
                equal.push_back(nums[i]);
            }
        }
        prev.insert(prev.end(), equal.begin(), equal.end());
        prev.insert(prev.end(), next.begin(), next.end());

        return prev;
    }
};
